\begin{notechapter}
\chapter{The Differential}
\label{ch:differential}
\chaptersubtitle{%
  The best affine approximation of a map between affine spaces}

\begin{chapterplan}
\begin{itemize}[nosep]
  \item Little-$o$ and tangency of maps; the differential $Df(a):V\to W$ of a
        map $f:U\subset\Aff\to\Bff$; uniqueness.
  \item Directional and partial derivatives; the Jacobian matrix;
        differentiability versus existence of partial derivatives; the $C^1$
        criterion.
  \item The tangent flat (an affine subspace) versus the tangent space (a vector
        space).
  \item Velocities of curves, and the identification $T_a\Aff = V$, which
        involves no choice.
  \item The differential of a scalar function is a covector, $\dd f_a\in V^*$.
  \item Worked examples: linear, affine and bilinear maps; the determinant; the
        inverse of a matrix.
\end{itemize}
\smallskip\noindent\textbf{Machine learning threads.}
\begin{itemize}[nosep]
  \item Jacobians of the usual layers.
  \item The linearization of a network around its parameters.
\end{itemize}
\end{chapterplan}
\end{notechapter}
