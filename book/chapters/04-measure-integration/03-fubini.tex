\chapter{Product Measures and the Fubini--Tonelli Theorem}
\label{ch:fubini}
\chaptersubtitle{%
  Integrating one variable at a time}

\begin{chapterintro}
A function of two variables can be integrated in $x$ and then in $y$,
in $y$ and then in $x$, or at once over the plane. This chapter builds
the measure that gives the third integral a meaning and proves that the
three agree: always for non-negative functions, and for integrable ones.
The same theorem says that a double series may be summed in either
order, that a joint probability is a marginal times a conditional, that
the expectation of a product of independent variables is the product of
the expectations, and that an average over a sample converges at the
rate $N^{-1/2}$ in every dimension. You will learn
\begin{itemize}[nosep]
  \item the product of two $\sigma$-algebras and of two measures, and
        the principle that a set is measured by its slices;
  \item Tonelli's theorem for non-negative functions and Fubini's
        theorem for integrable ones, with the examples where the
        exchange fails;
  \item independence as a product of laws, and the variance of a
        Monte Carlo average;
  \item convolution, the law of a sum of independent variables;
  \item marginal and conditional densities, and the risk as an iterated
        expectation.
\end{itemize}
\prerequisites{\cref{ch:measures,ch:lebesgue-integral}. Throughout,
$(X, \mathcal M, \mu)$ and $(Y, \mathcal N, \nu)$ are
\emph{$\sigma$-finite} measure spaces: each is a countable union of sets
of finite measure. Lebesgue measure, counting measure on a countable
set, and every probability measure are $\sigma$-finite.}
\end{chapterintro}

\section{Product $\sigma$-algebras}\label{fubini:sec:sigma}

\begin{definition}[Product $\sigma$-algebra]\label{fubini:def:product}
A \emph{measurable rectangle} is a set $A\times B\subset X\times Y$ with
$A\in\mathcal M$ and $B\in\mathcal N$. The \emph{product
$\sigma$-algebra} $\mathcal M\otimes\mathcal N$ is the $\sigma$-algebra
on $X\times Y$ generated by the measurable rectangles.
\end{definition}

The measurable rectangles form a $\pi$-system, since
$(A\times B)\cap(A'\times B') = (A\cap A')\times(B\cap B')$. That is
what will make \cref{measures:thm:pilambda} available.

\begin{proposition}\label{fubini:prop:components}
\begin{enumerate}[nosep]
  \item The projections of $X\times Y$ onto $X$ and $Y$ are measurable.
        A map $h = (h_1, h_2)$ from a measurable space to $X\times Y$ is
        measurable if and only if $h_1$ and $h_2$ are.
  \item $\mathcal B(\R^m)\otimes\mathcal B(\R^n) = \mathcal B(\R^{m+n})$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) The preimage of $A$ by the first projection is the rectangle
$A\times Y$; so the projections are measurable, and $h_1, h_2$ are if
$h$ is, as compositions. Conversely
$h^{-1}(A\times B) = h_1^{-1}(A)\cap h_2^{-1}(B)$, and it is enough to
test measurability on generators (\cref{measures:prop:measurable}).

(2) A half-open box of $\R^{m+n}$ is a product of two half-open boxes,
hence a measurable rectangle; the boxes generate
$\mathcal B(\R^{m+n})$ (\cref{measures:prop:boxes}), which is therefore
contained in the product. Conversely the projections of $\R^{m+n}$ are
continuous, hence Borel measurable, so every rectangle
$A\times B = \pi_1^{-1}(A)\cap\pi_2^{-1}(B)$ of Borel sets is a Borel set.
\end{proof}

\begin{definition}[Sections]\label{fubini:def:sections}
For $E\subset X\times Y$, $x\in X$ and $y\in Y$, the \emph{sections} of
$E$ are
\[
  E_x = \{y\in Y : (x, y)\in E\}\subset Y , \qquad
  E^y = \{x\in X : (x, y)\in E\}\subset X .
\]
\end{definition}

\begin{lemma}\label{fubini:lem:sections}
If $E\in\mathcal M\otimes\mathcal N$, then $E_x\in\mathcal N$ and
$E^y\in\mathcal M$ for all $x$ and $y$. If $f$ is a measurable function
on $X\times Y$, then $y\mapsto f(x, y)$ is measurable on $Y$ for every
$x$, and $x\mapsto f(x, y)$ is measurable on $X$ for every $y$.
\end{lemma}

\begin{proof}
Fix $x$. The sets $E$ with $E_x\in\mathcal N$ form a $\sigma$-algebra,
because $(E^c)_x = (E_x)^c$ and
$(\bigcup_nE_n)_x = \bigcup_n(E_n)_x$. It contains the rectangles:
$(A\times B)_x$ is $B$ or $\emptyset$. So it contains
$\mathcal M\otimes\mathcal N$. For a function,
$\{y : f(x, y)\in C\} = (f^{-1}(C))_x$.
\end{proof}

\section{The product measure}\label{fubini:sec:measure}

\begin{theorem}[Product measure]\label{fubini:thm:product}
For every $E\in\mathcal M\otimes\mathcal N$, the functions
$x\mapsto\nu(E_x)$ and $y\mapsto\mu(E^y)$ are measurable, and
\[
  (\mu\otimes\nu)(E) = \int_X\nu(E_x)\,\dd\mu(x)
  = \int_Y\mu(E^y)\,\dd\nu(y) .
\]
This defines a measure $\mu\otimes\nu$ on $\mathcal M\otimes\mathcal N$,
the \emph{product measure}. It is the only measure such that
$(\mu\otimes\nu)(A\times B) = \mu(A)\,\nu(B)$ for all measurable
rectangles, and it is $\sigma$-finite.
\end{theorem}

\begin{proof}
\emph{Measurability, $\nu$ finite.} Let $\mathcal L$ be the family of
sets $E\in\mathcal M\otimes\mathcal N$ for which $x\mapsto\nu(E_x)$ is
measurable. It contains the rectangles, since
$\nu((A\times B)_x) = \nu(B)1_A(x)$. It is a $\lambda$-system
(\cref{measures:def:pilambda}): it contains $X\times Y$; if $E\subset F$
are in $\mathcal L$ then $\nu((F\setminus E)_x) = \nu(F_x) - \nu(E_x)$,
a difference of finite measurable functions; and if $E_n$ increases to
$E$, then $\nu(E_x) = \lim_n\nu((E_n)_x)$ by
\cref{measures:thm:continuity}, a limit of measurable functions. By the
$\pi$--$\lambda$ theorem, $\mathcal L = \mathcal M\otimes\mathcal N$.

\emph{Measurability, $\nu$ $\sigma$-finite.} Let $Y_k$ increase to $Y$
with $\nu(Y_k) < \infty$. The measures $\nu_k(B) = \nu(B\cap Y_k)$ are
finite, and $\nu(E_x) = \lim_k\nu_k(E_x)$: again a limit of measurable
functions.

\emph{A measure.} Put $\rho(E) = \int\nu(E_x)\,\dd\mu(x)$. If the $E_n$
are disjoint with union $E$, their sections at $x$ are disjoint with
union $E_x$, so $\nu(E_x) = \sum_n\nu((E_n)_x)$, and
$\rho(E) = \sum_n\rho(E_n)$ by \cref{lebesgue-integral:cor:additive}.
So $\rho$ is a measure, and
$\rho(A\times B) = \int\nu(B)1_A\,\dd\mu = \mu(A)\nu(B)$.

\emph{Uniqueness and symmetry.} Let $X_k$ increase to $X$ with
$\mu(X_k) < \infty$. The rectangles $X_k\times Y_k$ increase to
$X\times Y$ and have finite measure for any measure that is
$\mu(A)\nu(B)$ on rectangles; so there is only one such measure
(\cref{measures:cor:uniqueness}). Exchanging the roles of $X$ and $Y$
in what precedes, $y\mapsto\mu(E^y)$ is measurable and
$\rho'(E) = \int\mu(E^y)\,\dd\nu(y)$ is another such measure. Hence
$\rho' = \rho$.
\end{proof}

\begin{keyidea}[A set is measured by its slices]
The measure of $E$ is the integral of the measures of its sections, and
it does not matter in which direction $E$ is sliced
(\cref{fubini:fig:slices}). Two sets whose sections have the same
measure at every $x$ have the same measure: this is Cavalieri's
principle.
\end{keyidea}

\begin{corollary}[Lebesgue measure is a product]\label{fubini:cor:lebesgue}
$\lambda^m\otimes\lambda^n = \lambda^{m+n}$ on
$\mathcal B(\R^{m+n})$, where $\lambda^k$ is the Lebesgue measure of
$\R^k$.
\end{corollary}

\begin{proof}
Both are measures on $\mathcal B(\R^{m+n})$
(\cref{fubini:prop:components}) which give a half-open box the product
of the lengths of its sides. Apply the uniqueness in
\cref{measures:thm:lebesgue}.
\end{proof}

\begin{proposition}[The integral is the area under the graph]
\label{fubini:prop:layer}
For a measurable $f:X\to[0, \infty]$,
\[
  \int_Xf\,\dd\mu
  = (\mu\otimes\lambda)\bigl\{(x, t) : 0\leq t < f(x)\bigr\}
  = \int_0^\infty\mu\{f > t\}\,\dd t .
\]
In particular $\E[Z] = \int_0^\infty P(Z > t)\,\dd t$ for a random
variable $Z\geq0$.
\end{proposition}

\begin{proof}
The set $E = \{(x, t) : 0\leq t < f(x)\}$ is measurable: $t < f(x)$ if
and only if $t\leq q < f(x)$ for some rational $q$, so $E$ is the
countable union of the rectangles $\{f > q\}\times[0, q]$. Its sections
are $E_x = [0, f(x))$, of length $f(x)$, and $E^t = \{f > t\}$. Apply
\cref{fubini:thm:product}.
\end{proof}

\begin{pitfall}[Without $\sigma$-finiteness]
On $[0, 1]$ take $\mu = \lambda$ and let $\nu$ be the counting measure
on the Borel sets, which is not $\sigma$-finite. The diagonal
$D = \{(x, x)\}$ is closed, hence measurable. Its sections are single
points: $\nu(D_x) = 1$ and $\lambda(D^y) = 0$. So
$\int\nu(D_x)\,\dd\lambda = 1$ while $\int\lambda(D^y)\,\dd\nu = 0$.
\end{pitfall}

For three or more factors the product is formed step by step, and
$(\mu_1\otimes\mu_2)\otimes\mu_3 = \mu_1\otimes(\mu_2\otimes\mu_3)$,
since both give $\mu_1(A_1)\mu_2(A_2)\mu_3(A_3)$ to a product of three
sets. We write $\mu^{\otimes N}$ for the product of $N$ copies of $\mu$.

\section{The theorems of Tonelli and Fubini}\label{fubini:sec:theorems}

\begin{theorem}[Tonelli]\label{fubini:thm:tonelli}
Let $f:X\times Y\to[0, \infty]$ be measurable. Then
$x\mapsto\int_Yf(x, y)\,\dd\nu(y)$ and
$y\mapsto\int_Xf(x, y)\,\dd\mu(x)$ are measurable, and
\[
  \int_{X\times Y}f\,\dd(\mu\otimes\nu)
  = \int_X\Bigl(\int_Yf(x, y)\,\dd\nu(y)\Bigr)\dd\mu(x)
  = \int_Y\Bigl(\int_Xf(x, y)\,\dd\mu(x)\Bigr)\dd\nu(y) .
\]
\end{theorem}

\begin{proof}
For $f = 1_E$ the inner integral is $\nu(E_x)$, and the statement is
\cref{fubini:thm:product}. By linearity it holds for simple functions.
For a general $f$ let $s_n$ be simple functions increasing to $f$
(\cref{lebesgue-integral:lem:dyadic}). For every $x$,
$\int s_n(x, y)\,\dd\nu(y)$ increases to $\int f(x, y)\,\dd\nu(y)$ by
monotone convergence; so this function of $x$ is measurable, as a
limit, and by monotone convergence twice more, on $X$ and on
$X\times Y$,
\begin{align*}
  \int\Bigl(\int f(x, y)\,\dd\nu(y)\Bigr)\dd\mu(x)
  &= \lim_n\int\Bigl(\int s_n(x, y)\,\dd\nu(y)\Bigr)\dd\mu(x) \\
  &= \lim_n\int s_n\,\dd(\mu\otimes\nu) = \int f\,\dd(\mu\otimes\nu) .
\end{align*}
The other order is the same.
\end{proof}

\begin{theorem}[Fubini]\label{fubini:thm:fubini}
Let $f:X\times Y\to\R$ be integrable for $\mu\otimes\nu$. Then
\begin{enumerate}[nosep]
  \item for almost every $x$, the function $y\mapsto f(x, y)$ is
        integrable on $Y$;
  \item the function $x\mapsto\int_Yf(x, y)\,\dd\nu(y)$, defined almost
        everywhere, is integrable on $X$;
  \item $\displaystyle\int_{X\times Y}f\,\dd(\mu\otimes\nu)
        = \int_X\Bigl(\int_Yf(x, y)\,\dd\nu(y)\Bigr)\dd\mu(x)$.
\end{enumerate}
The same holds with the roles of $x$ and $y$ exchanged, so the two
iterated integrals are equal.
\end{theorem}

\begin{proof}
By Tonelli's theorem applied to $\abs f$, the function
$G(x) = \int\abs{f(x, y)}\,\dd\nu(y)$ is measurable with
$\int G\,\dd\mu = \int\abs f\,\dd(\mu\otimes\nu) < \infty$. So $G$ is
finite outside a null set $N$ (\cref{lebesgue-integral:prop:markov}),
which is (1). Apply Tonelli's theorem to $f^+$ and $f^-$: the functions
$F^\pm(x) = \int f^\pm(x, y)\,\dd\nu(y)$ are measurable, at most $G$,
and have integrals $\int f^\pm\,\dd(\mu\otimes\nu)$. Outside $N$ the
inner integral of $f$ is $F^+ - F^-$; setting it to $0$ on $N$ gives a
measurable function, integrable since it is bounded by $G$, and
subtracting the two identities gives (3).
\end{proof}

\begin{keyidea}[How the two theorems are used]
To exchange two integrals, or an integral and a sum:
\begin{enumerate}[nosep]
  \item if the integrand is non-negative, exchange (Tonelli); the
        common value may be $\infty$;
  \item otherwise compute, or bound, an iterated integral of
        $\abs f$, in whichever order is easier. If it is finite, $f$ is
        integrable on the product (Tonelli), and the exchange is
        allowed (Fubini).
\end{enumerate}
\end{keyidea}

\begin{example}[The Gaussian integral]\label{fubini:ex:gauss}
Let $I = \int_0^\infty e^{-x^2}\dd x$. For $x > 0$ the substitution
$y = xs$ gives $I = \int_0^\infty xe^{-x^2s^2}\,\dd s$, so
\[
  I^2 = \int_0^\infty e^{-x^2}\Bigl(\int_0^\infty
        xe^{-x^2s^2}\,\dd s\Bigr)\dd x
  = \int_0^\infty\Bigl(\int_0^\infty xe^{-x^2(1 + s^2)}\,\dd x\Bigr)\dd s
  = \int_0^\infty\frac{\dd s}{2(1 + s^2)} = \frac\pi4 ,
\]
the exchange being Tonelli's theorem for a non-negative function.
Hence $\int_\R e^{-x^2}\dd x = \sqrt\pi$, and the density
$\varphi(x) = e^{-x^2/2}/\sqrt{2\pi}$ has integral $1$.
\end{example}

\begin{example}[Double series]\label{fubini:ex:series}
For the counting measures on two countable sets, a function is a doubly
indexed family $a_{mn}$ and the product measure is the counting measure
on the pairs. Tonelli: if $a_{mn}\geq0$, then
$\sum_m\sum_na_{mn} = \sum_n\sum_ma_{mn}$. Fubini: the same holds if
$\sum_m\sum_n\abs{a_{mn}} < \infty$, and the family may then be summed
in any order.
\end{example}

\begin{pitfall}[When the exchange fails]
Let $a_{mn} = 1$ if $n = m$, $-1$ if $n = m + 1$, and $0$ otherwise
(\cref{fubini:fig:array}). Every row has sum $0$, so
$\sum_m\sum_na_{mn} = 0$. The first column has sum $1$ and the others
$0$, so $\sum_n\sum_ma_{mn} = 1$. Here $\sum\abs{a_{mn}} = \infty$. A
continuous version: on $(0, 1)^2$,
\[
  f(x, y) = \frac{x^2 - y^2}{(x^2 + y^2)^2} , \qquad
  \int_0^1\Bigl(\int_0^1f\,\dd y\Bigr)\dd x = \frac\pi4 , \qquad
  \int_0^1\Bigl(\int_0^1f\,\dd x\Bigr)\dd y = -\frac\pi4
\]
(\cref{fubini:exr:3}). Both iterated integrals exist; the function is
not integrable on the square.
\end{pitfall}

\section{Independence}\label{fubini:sec:independence}

\begin{definition}[Independence]\label{fubini:def:independent}
Random variables $Z_1,\dots,Z_N$ on a probability space, with values in
measurable spaces $X_1,\dots,X_N$, are \emph{independent} if the law of
$(Z_1,\dots,Z_N)$ is the product of their laws:
\[
  P_{(Z_1,\dots,Z_N)} = P_{Z_1}\otimes\cdots\otimes P_{Z_N} ,
  \quad\text{that is,}\quad
  P(Z_1\in A_1,\dots,Z_N\in A_N) = \prod_{i=1}^NP(Z_i\in A_i)
\]
for all measurable $A_i\subset X_i$.
\end{definition}

The two forms are equivalent because a measure on the product is
determined by its values on rectangles. Independent variables with given
laws $\mu_i$ always exist: on $\Omega = X_1\times\cdots\times X_N$ with
$P = \mu_1\otimes\cdots\otimes\mu_N$, take the coordinates. Measurable
functions $g_i(Z_i)$ of independent variables are independent, since
$\{g_i(Z_i)\in A\} = \{Z_i\in g_i^{-1}(A)\}$.

\begin{proposition}\label{fubini:prop:product}
Let $Z_1, Z_2$ be independent and $g, h$ real measurable functions. If
$g(Z_1)$ and $h(Z_2)$ are integrable, so is their product, and
\[
  \E\bigl[g(Z_1)h(Z_2)\bigr] = \E\bigl[g(Z_1)\bigr]\,\E\bigl[h(Z_2)\bigr] .
\]
The formula also holds, in $[0, \infty]$, if $g, h\geq0$.
\end{proposition}

\begin{proof}
By the transfer theorem (\cref{lebesgue-integral:thm:transfer}),
independence, and Tonelli's theorem,
\[
  \E\bigl[\abs{g(Z_1)h(Z_2)}\bigr]
  = \int\abs{g(x)}\abs{h(y)}\,\dd(P_{Z_1}\otimes P_{Z_2})
  = \int\abs g\,\dd P_{Z_1}\int\abs h\,\dd P_{Z_2} .
\]
If this is finite, the same computation without absolute values is
Fubini's theorem.
\end{proof}

For a real random variable with $\E[Z^2] < \infty$, which is then
integrable because $\abs z\leq1 + z^2$, the \emph{variance} is
$\Var(Z) = \E[(Z - \E Z)^2] = \E[Z^2] - (\E Z)^2$. If $Z_1, Z_2$ are
independent,
$\E[(Z_1 - \E Z_1)(Z_2 - \E Z_2)] = 0$ by the proposition, so
$\Var(Z_1 + Z_2) = \Var(Z_1) + \Var(Z_2)$.

\begin{theorem}[Monte Carlo]\label{fubini:thm:mc}
Let $Z_1,\dots,Z_N$ be independent with the same law, and $g$ a real
measurable function with $\E[g(Z_1)^2] < \infty$. Put
$m = \E[g(Z_1)]$, $\sigma^2 = \Var(g(Z_1))$ and
$\hat g_N = \frac1N\sum_{i=1}^Ng(Z_i)$. Then
\[
  \E[\hat g_N] = m , \qquad \Var(\hat g_N) = \frac{\sigma^2}N , \qquad
  P\bigl(\abs{\hat g_N - m}\geq\epsilon\bigr)
  \leq\frac{\sigma^2}{N\epsilon^2}\quad(\epsilon > 0) .
\]
\end{theorem}

\begin{proof}
The mean is linearity. Expanding the square,
\[
  \Var\Bigl(\sum_ig(Z_i)\Bigr) = \sum_i\Var\bigl(g(Z_i)\bigr)
  + \sum_{i\neq j}\E\bigl[(g(Z_i) - m)(g(Z_j) - m)\bigr] ,
\]
and each
cross term vanishes by \cref{fubini:prop:product}, since $Z_i$ and
$Z_j$ are independent for $i\neq j$; so the
variance of the sum is $N\sigma^2$, and dividing by $N$ divides the
variance by $N^2$. The last statement is Markov's inequality
(\cref{lebesgue-integral:prop:markov}) for $(\hat g_N - m)^2$ and
$c = \epsilon^2$.
\end{proof}

\begin{intuition}
The error of an average of $N$ independent samples is of order
$\sigma/\sqrt N$. Nothing in this statement mentions the space where the
samples live. A quadrature rule on a grid needs a number of points that
grows exponentially with the dimension; an average over samples does
not, which is why integrals in high dimension are computed by sampling.
\end{intuition}

\section{Convolution}\label{fubini:sec:convolution}

\begin{lemma}[Integrating against a density]\label{fubini:lem:density}
Let $p:X\to[0, \infty]$ be measurable and
$\rho(A) = \int_Ap\,\dd\mu$, the measure with \emph{density} $p$ with
respect to $\mu$. Then $\int g\,\dd\rho = \int gp\,\dd\mu$ for every
measurable $g\geq0$, and for every $g$ such that $gp$ is
$\mu$-integrable.
\end{lemma}

\begin{proof}
For $g = 1_A$ it is the definition; then simple functions,
non-negative functions by monotone convergence, and differences, as in
the proof of \cref{lebesgue-integral:thm:transfer}.
\end{proof}

\begin{definition}[Convolution]\label{fubini:def:convolution}
The \emph{convolution} of two finite measures $\mu, \nu$ on
$\mathcal B(\R^n)$ is the push-forward of $\mu\otimes\nu$ by the sum
$(x, y)\mapsto x + y$:
\[
  (\mu*\nu)(A) = (\mu\otimes\nu)\bigl\{(x, y) : x + y\in A\bigr\} .
\]
\end{definition}

If $Z_1$ and $Z_2$ are independent with laws $\mu$ and $\nu$, then
$\mu*\nu$ is the law of $Z_1 + Z_2$.

\begin{theorem}\label{fubini:thm:convolution}
Let $\mu$ be a finite measure on $\mathcal B(\R^n)$ and let $\nu$ be a
finite measure with a density $g$ with respect to Lebesgue measure.
Then $\mu*\nu$ has the
density
\[
  (\mu*g)(z) = \int g(z - x)\,\dd\mu(x) .
\]
If $\mu$ has a density $f$, this is
$(f*g)(z) = \int f(x)\,g(z - x)\,\dd x = \int f(z - y)\,g(y)\,\dd y$.
\end{theorem}

\begin{proof}
Let $A$ be a Borel set. By Tonelli's theorem and
\cref{fubini:lem:density},
\[
  (\mu*\nu)(A) = \int\Bigl(\int1_A(x + y)\,g(y)\,\dd y\Bigr)\dd\mu(x)
  = \int\Bigl(\int1_A(z)\,g(z - x)\,\dd z\Bigr)\dd\mu(x) ,
\]
where the inner integral was transported by the translation
$y\mapsto x + y$, which preserves Lebesgue measure
(\cref{measures:thm:translation,lebesgue-integral:thm:transfer}). The
function $(x, z)\mapsto g(z - x)$ is measurable, as a composition, so
Tonelli's theorem applies again:
$(\mu*\nu)(A) = \int_A\bigl(\int g(z - x)\,\dd\mu(x)\bigr)\dd z$. The last
sentence is \cref{fubini:lem:density} and the change $y = z - x$.
\end{proof}

\begin{example}[Sum of two uniform variables]\label{fubini:ex:uniform}
Let $f = g = 1_{[0, 1]}$. Then $(f*g)(z)$ is the length of
$[0, 1]\cap[z - 1, z]$: it is $z$ for $0\leq z\leq1$, $2 - z$ for
$1\leq z\leq2$, and $0$ elsewhere (\cref{fubini:fig:convolution}). The
sum of two independent uniform variables has a triangular density.
\end{example}

\begin{mlthread}[Machine learning: adding noise is a convolution]
\begin{itemize}[nosep]
  \item \emph{Smoothing.} If $x$ is drawn from any distribution $\mu$
        on $\R^n$ and $\varepsilon$ is independent noise with a Gaussian
        density $\varphi_\sigma$, then by \cref{fubini:thm:convolution}
        the law of $x + \varepsilon$ has the density
        $z\mapsto\int\varphi_\sigma(z - x)\,\dd\mu(x)$, even when $\mu$
        has none, like the model of \cref{measures:fig:generative}.
        Diffusion models and denoising score matching start from this
        fact: the noised data always have a smooth positive density,
        whose logarithm can be differentiated.
  \item \emph{Kernel density estimation.} For the empirical measure of
        a sample the same formula reads
        $\frac1N\sum_ig(z - x_i)$: a bump around each data point.
  \item \emph{Variances add.} The convolution of centred Gaussian
        densities of variances $s^2$ and $t^2$ is Gaussian of variance
        $s^2 + t^2$ (\cref{fubini:exr:7}). Adding noise in many small
        steps or in one step gives the same law.
\end{itemize}
\end{mlthread}

\section{Marginals, conditionals and the risk}\label{fubini:sec:ml}

Let $P$ be a probability measure on $X\times Y$, the distribution of a
pair $(x, y)$: an input and a label, or an observation and a latent
variable. Assume that it has a density $p(x, y)$ with respect to
$\mu\otimes\nu$. For labels in a finite set $\nu$ is the counting
measure, and the integrals in $y$ below are sums.

\begin{proposition}[Marginal and conditional densities]
\label{fubini:prop:marginal}
The law of $x$, the \emph{marginal} $P_X$, has the density
\[
  p_X(x) = \int_Yp(x, y)\,\dd\nu(y)
\]
with respect to $\mu$. Where $0 < p_X(x) < \infty$, which holds for
$P_X$-almost every $x$, put $p(y\mid x) = p(x, y)/p_X(x)$, a probability
density on $Y$; elsewhere put $p(y\mid x) = 0$. Then for every
measurable $h\geq0$, and for every
$P$-integrable $h$,
\[
  \int h\,\dd P
  = \int_X\Bigl(\int_Yh(x, y)\,p(y\mid x)\,\dd\nu(y)\Bigr)
    p_X(x)\,\dd\mu(x) .
\]
\end{proposition}

\begin{proof}
By \cref{fubini:lem:density} and Tonelli's theorem,
$P_X(A) = P(A\times Y) = \int_A\bigl(\int p(x, y)\,\dd\nu(y)\bigr)\dd\mu(x)$.
Since $\int p_X\,\dd\mu = 1$, $p_X$ is finite $\mu$-almost everywhere,
and $P_X\{p_X = 0\} = \int_{\{p_X = 0\}}p_X\,\dd\mu = 0$. The formula is
Tonelli's theorem for $hp$: where $0 < p_X(x) < \infty$,
$p(x, y) = p(y\mid x)\,p_X(x)$; where $p_X(x) = 0$, $p(x, y) = 0$ for
$\nu$-almost every $y$ and both inner integrals vanish; and
$\{p_X = \infty\}$ is $\mu$-null. For integrable $h$ it is Fubini's
theorem.
\end{proof}

\begin{keyidea}[The risk is an iterated expectation]
$\int h\,\dd P = \E_x\bigl[\E_{y\mid x}[h(x, y)]\bigr]$: average over
the labels given the input, then over the inputs. So an expected loss
$\E[\ell(\hat y(x), y)]$ is minimized over all predictors $\hat y$ by
minimizing, for each $x$ separately, the inner expectation
$\E_{y\mid x}[\ell(\hat y, y)]$.
\end{keyidea}

\begin{proposition}[Regression function]\label{fubini:prop:regression}
Let $Y = \R$ and $\int y^2\,\dd P < \infty$, and let
$m(x) = \int y\,p(y\mid x)\,\dd\nu(y)$. For every measurable
$h:X\to\R$ with $\int h(x)^2\,\dd P < \infty$,
\[
  \int\bigl(y - h(x)\bigr)^2\dd P
  = \int\bigl(y - m(x)\bigr)^2\dd P
  + \int\bigl(h(x) - m(x)\bigr)^2\dd P_X .
\]
\end{proposition}

\begin{proof}
By \cref{fubini:prop:marginal}, $v(x) = \int y^2p(y\mid x)\,\dd\nu(y)$ is
finite for $P_X$-almost every $x$. For such an $x$ the measure
$p(y\mid x)\,\dd\nu(y)$ is a probability under which $y$ is square
integrable, so $m(x)$ is defined and finite with $m(x)^2\leq v(x)$ by
Cauchy--Schwarz, and
$0\leq\int(y - m(x))^2p(y\mid x)\,\dd\nu(y) = v(x) - m(x)^2$. So
$\int m^2\,\dd P_X\leq\int v\,\dd P_X = \int y^2\,\dd P$ is finite. Write
$y - h = (y - m) + (m - h)$ and expand the square. The cross term
$(y - m(x))(m(x) - h(x))$ is integrable, being at most half the sum of
two integrable squares, and its integral is
\[
  \int_X\bigl(m(x) - h(x)\bigr)
  \Bigl(\int_Y\bigl(y - m(x)\bigr)p(y\mid x)\,\dd\nu(y)\Bigr)
  p_X(x)\,\dd\mu(x) = 0 ,
\]
because the inner integral vanishes by the definition of $m$.
\end{proof}

So the predictor with the smallest squared risk is the conditional mean
$m$, and its risk, the first term, is the part of the risk that no
predictor can remove. For a classification loss the same principle
gives the label that maximizes $p(y\mid x)$ (\cref{fubini:exr:9}).
Conditional expectations without densities are constructed in
\cref{ch:lp-spaces}.

\begin{mlthread}[Machine learning: sums, products and samples]
\begin{itemize}[nosep]
  \item \emph{Marginalization.} A latent variable model defines
        $p_\theta(x, z) = p_\theta(x\mid z)\,p(z)$ and is fitted through
        the marginal $p_\theta(x) = \int p_\theta(x\mid z)p(z)\,\dd\nu(z)$:
        a finite sum for a mixture, an integral for a continuous latent
        space. The ``sum rule'' and the ``product rule'' of probability
        are \cref{fubini:prop:marginal}.
  \item \emph{Ancestral sampling.} Drawing $z$ from $p(z)$ and then $x$
        from $p_\theta(x\mid z)$ gives a sample of the joint law;
        discarding $z$ gives a sample of the marginal, whose density may
        be impossible to compute.
  \item \emph{Mini-batches.} With $B$ independent examples the
        mini-batch gradient has mean the gradient of the risk, when the
        exchange of \cref{ch:lebesgue-integral} is allowed, and each of
        its components has variance $\sigma^2/B$
        (\cref{fubini:thm:mc}): four times the batch for half the
        noise.
  \item \emph{Nested expectations.} An objective such as
        $\E_x\bigl[(\E_{y\mid x}[g])^2\bigr]$ is not the expectation of
        $g^2$. Estimating the square of an inner expectation without
        bias takes two independent inner samples
        (\cref{fubini:exr:8}).
\end{itemize}
\end{mlthread}

\section{Where the chapter leads}\label{fubini:sec:outlook}

Integrals over products reduce to integrals over the factors, in any
order, for non-negative or integrable functions. \Cref{ch:lp-spaces}
studies the spaces of functions with $\int\abs f^p\,\dd\mu < \infty$:
the square-integrable ones form an inner product space in which
\cref{fubini:prop:regression} is an orthogonal projection, and the
existence of densities becomes a theorem.
\Cref{ch:change-of-variables} computes integrals over $\R^n$ in
curvilinear coordinates, with the present chapter as its main tool.

\paragraph{Sources.} Loomis and Sternberg, \emph{Advanced Calculus},
\S8.12 (``Successive integration''), prove that the integral of a
function which is contented relative to a decomposition
$\R^n = \R^k\times\R^l$ (their Definition~12.1) is the iterated integral
(their equation~(12.1)). Hubbard and
Hubbard state Fubini's theorem for Riemann integrals (Theorem~4.5.10,
proved in their Appendix~A17), use it to compute probabilities
(Example~4.5.8), and state it for the Lebesgue integral with a converse
that plays the role of Tonelli's theorem (Theorem~4.11.21). The
construction of the product measure from the measures of the
sections, with the $\pi$--$\lambda$ theorem, is that of P.~Billingsley,
\emph{Probability and Measure}, 3rd ed., Wiley, 1995, Section~18
(Theorems~18.1 to~18.3). M.~Capi\'nski and E.~Kopp, \emph{Measure,
Integral and Probability}, 2nd ed., Springer, 2004, Chapter~6, carry
it out for finite measures, with the monotone class theorem (their
Lemma~6.5) in place of the $\pi$--$\lambda$ theorem; Fubini's theorem
is their Theorem~6.10.

\paragraph{Notebook.} The numerical claims of this chapter are checked
with PyTorch in \notebook{18-fubini.ipynb}.

\begin{chaptersummary}
\begin{itemize}[nosep]
  \item $\mathcal M\otimes\mathcal N$ is generated by the rectangles
        $A\times B$; sections of measurable sets and functions are
        measurable; $\mathcal B(\R^m)\otimes\mathcal B(\R^n)
        = \mathcal B(\R^{m+n})$.
  \item For $\sigma$-finite measures there is one measure with
        $(\mu\otimes\nu)(A\times B) = \mu(A)\nu(B)$, and
        $(\mu\otimes\nu)(E) = \int\nu(E_x)\,\dd\mu = \int\mu(E^y)\,\dd\nu$.
        Lebesgue measure on $\R^{m+n}$ is
        $\lambda^m\otimes\lambda^n$.
  \item Tonelli: for $f\geq0$ the two iterated integrals and the
        integral on the product are equal. Fubini: the same for
        integrable $f$. Check integrability with Tonelli on $\abs f$.
  \item $\int f\,\dd\mu = \int_0^\infty\mu\{f > t\}\,\dd t$.
  \item Independent variables have a product law;
        $\E[g(Z_1)h(Z_2)] = \E[g(Z_1)]\E[h(Z_2)]$; variances add; an
        average of $N$ independent samples has variance $\sigma^2/N$.
  \item The law of a sum of independent variables is the convolution of
        their laws; with a density $g$ it has density
        $\int g(z - x)\,\dd\mu(x)$.
  \item A joint density is a marginal times a conditional, and the risk
        is an iterated expectation; the conditional mean minimizes the
        squared risk.
\end{itemize}
\end{chaptersummary}
