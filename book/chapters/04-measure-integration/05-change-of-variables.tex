\chapter{Change of Variables}
\label{ch:change-of-variables}
\chaptersubtitle{%
  How densities transform}

\begin{chapterintro}
A linear map multiplies volumes by the absolute value of its
determinant (\cref{measures:thm:linear}). A diffeomorphism does the
same thing point by point, with the determinant of its differential:
that is the change of variables theorem. It says how integrals are
computed in curvilinear coordinates, and how a probability density
changes when the variable is transformed: by composition \emph{and}
division by a Jacobian determinant. A density is therefore not a
function on the space. The same computation with a rectangular Jacobian
matrix gives lengths, areas and volumes on submanifolds through the Gram
determinant $\sqrt{\det G}$, the volume of a Riemannian manifold, and
the surprising geometry of high-dimensional balls and Gaussians. You
will learn
\begin{itemize}[nosep]
  \item the change of variables theorem for $C^1$ diffeomorphisms, with
        its proof;
  \item the density of the push-forward of a measure by a
        diffeomorphism, and why modes are not invariant;
  \item polar coordinates in the plane and in $\R^n$;
  \item the volume measure of a submanifold and its independence of the
        parametrization;
  \item the area of spheres, the volume of balls, and where the mass of
        a high-dimensional Gaussian sits;
  \item normalizing flows, and the prior that is the same in all
        parametrizations.
\end{itemize}
\prerequisites{\cref{ch:measures,ch:lebesgue-integral,ch:fubini}, the
density \cref{lp-spaces:thm:dense}, diffeomorphisms and the
inverse function theorem (\cref{ch:inverse-implicit}), and for the last
sections the parametrizations of \cref{ch:submanifolds}. Integrals over
subsets of $\R^n$ are for Lebesgue measure $\lambda$, written $\dd x$.}
\end{chapterintro}

\section{The theorem}\label{change-of-variables:sec:theorem}

Throughout, $U$ and $V$ are open subsets of $\R^n$ and $g:U\to V$ is a
diffeomorphism of class $C^1$ (\cref{inverse-implicit:def:diffeo}). By
the inverse function theorem, an injective map of class $C^1$ on $U$
whose differential is invertible at every point is a diffeomorphism
onto its image, which is open.

\begin{theorem}[Change of variables]\label{change-of-variables:thm:cov}
For every measurable $f:V\to[0, \infty]$,
\[
  \int_Vf(y)\,\dd y = \int_Uf\bigl(g(x)\bigr)\,\abs{\det Dg(x)}\,\dd x .
\]
A measurable $f:V\to\R$ is integrable on $V$ if and only if
$(f\circ g)\abs{\det Dg}$ is integrable on $U$, and the formula then
holds.
\end{theorem}

\begin{intuition}
Substitute $y = g(x)$ and replace $\dd y$ by
$\abs{\det Dg(x)}\,\dd x$. Near a point $a$ the map is close to the
affine map $x\mapsto g(a) + Dg(a)(x - a)$, which multiplies the volume
of a small cube by $\abs{\det Dg(a)}$
(\cref{change-of-variables:fig:polar}). The proof makes ``close to''
precise on cubes that shrink.
\end{intuition}

On $\R^n$ we use the norm $\norm x_\infty = \max_i\abs{x_i}$, whose
balls are cubes, and the corresponding operator norm. A \emph{closed
cube} is a set $\{x : \norm{x - c}_\infty\leq r\}$.

\begin{lemma}\label{change-of-variables:lem:cube}
Let $h:U\to\R^n$ be of class $C^1$ and $Q\subset U$ a closed cube. If
$\norm{Dh(x)}\leq K$ for all $x\in Q$, then
$\lambda(h(Q))\leq K^n\lambda(Q)$.
\end{lemma}

\begin{proof}
Let $c$ be the centre of $Q$ and $r$ its half-side. For $x\in Q$ the
segment $[c, x]$ lies in $Q$, and by the mean value inequality
(\cref{contractions-newton:thm:mvi})
$\norm{h(x) - h(c)}_\infty\leq Kr$. So $h(Q)$, which is compact, lies in
a cube of half-side $Kr$.
\end{proof}

\begin{lemma}\label{change-of-variables:lem:local}
For every closed cube $Q\subset U$,
$\lambda(g(Q))\leq\int_Q\abs{\det Dg}\,\dd\lambda$.
\end{lemma}

\begin{proof}
Let $\epsilon > 0$. There is a $\delta > 0$ such that
\[
  \norm{Dg(a)^{-1}Dg(x)}\leq1 + \epsilon
  \qquad\text{whenever }a, x\in Q,\ \norm{x - a}_\infty\leq\delta .
\]
Otherwise there would be $a_k, x_k\in Q$ with
$\norm{x_k - a_k}_\infty\to0$ violating it. Since $Q$ is compact
(\cref{norms-banach:thm:bw}) a subsequence of $(a_k)$ converges to some
$a\in Q$, and so does $(x_k)$. But $x\mapsto Dg(x)$ and
$x\mapsto Dg(x)^{-1} = D(g^{-1})(g(x))$ are continuous, so
$Dg(a_k)^{-1}Dg(x_k)$ tends to the identity along the subsequence: a
contradiction.

Cut $Q$ into finitely many closed cubes $Q_i$ of side at most $\delta$.
In each, let $a_i$ be a point where the continuous function
$\abs{\det Dg}$ is smallest (\cref{norms-banach:thm:extreme}) and put
$T = Dg(a_i)$. The map $h = T^{-1}\circ g$ has
$\norm{Dh(x)} = \norm{Dg(a_i)^{-1}Dg(x)}\leq1 + \epsilon$ on $Q_i$. By
\cref{measures:thm:linear,change-of-variables:lem:cube},
\[
  \lambda\bigl(g(Q_i)\bigr) = \abs{\det T}\,\lambda\bigl(h(Q_i)\bigr)
  \leq(1 + \epsilon)^n\abs{\det Dg(a_i)}\,\lambda(Q_i)
  \leq(1 + \epsilon)^n\int_{Q_i}\abs{\det Dg}\,\dd\lambda .
\]
The $Q_i$ overlap only along faces, which are null sets. Adding up
gives $\lambda(g(Q))\leq(1 + \epsilon)^n\int_Q\abs{\det Dg}\,\dd\lambda$,
for every $\epsilon > 0$.
\end{proof}

\begin{lemma}\label{change-of-variables:lem:borel}
For every Borel set $A\subset U$, the set $g(A)$ is a Borel set and
$\lambda(g(A))\leq\int_A\abs{\det Dg}\,\dd\lambda$.
\end{lemma}

\begin{proof}
Since $g^{-1}$ is continuous, $g(A) = (g^{-1})^{-1}(A)$ is a Borel set,
and $\nu(A) = \lambda(g(A))$ is a measure on $\mathcal B(U)$, as is
$\rho(A) = \int_A\abs{\det Dg}\,\dd\lambda$. We must show
$\nu\leq\rho$.

A \emph{dyadic cube} is a half-open box
$\prod_i[m_i2^{-j}, (m_i + 1)2^{-j})$ with integers $m_i$ and $j\geq0$;
two dyadic cubes are nested or disjoint. Let $K$ be a dyadic cube whose
closure $\bar K$ lies in $U$. By \cref{change-of-variables:lem:local},
$\nu(K)\leq\nu(\bar K)\leq\rho(\bar K) = \rho(K) < \infty$, and the same
holds for every dyadic cube inside $K$. So $\int s\,\dd\nu
\leq\int s\,\dd\rho$ for every function $s = \sum_ic_i1_{C_i}$ with
$c_i\geq0$ and $C_i$ dyadic cubes inside $K$. Now let $B\subset K$ be a
Borel set and $\epsilon > 0$. The measure $\mu(E) = (\nu + \rho)(E\cap K)$
is finite, so by \cref{lp-spaces:thm:dense} there is a combination
$\varphi$ of indicators of dyadic boxes with
$\int\abs{1_B - \varphi}\,\dd\mu < \epsilon$. Cutting these boxes into
dyadic cubes of one size, smaller than $K$, we may assume that they are
disjoint cubes and that each lies inside $K$ or outside it; those
outside can be dropped. Then
$s = \max(\varphi, 0)$ is a function of the kind above, and
$\abs{1_B - s}\leq\abs{1_B - \varphi}$. Hence
\[
  \nu(B)\leq\int s\,\dd\nu + \epsilon\leq\int s\,\dd\rho + \epsilon
  \leq\rho(B) + 2\epsilon .
\]

Finally, every point of $U$ lies in a dyadic cube with closure in $U$,
and in a largest one, since the sides are at most $1$. These largest
cubes are disjoint, countably many, and cover $U$. Apply the above to
the pieces of $A$ in each of them and add.
\end{proof}

\begin{proof}[Proof of \cref{change-of-variables:thm:cov}]
Let $f = 1_B$ for a Borel set $B\subset V$. With $A = g^{-1}(B)$,
\cref{change-of-variables:lem:borel} reads
$\int_V1_B\,\dd\lambda\leq\int_U(1_B\circ g)\abs{\det Dg}\,\dd\lambda$. By
linearity and monotone convergence along simple functions,
\[
  \int_Vf\,\dd\lambda\leq\int_U(f\circ g)\,\abs{\det Dg}\,\dd\lambda
  \qquad\text{for every measurable }f\geq0 .
\]
The same inequality holds for the diffeomorphism $g^{-1}:V\to U$ and the
function $F = (f\circ g)\abs{\det Dg}$ on $U$:
\[
  \int_U(f\circ g)\,\abs{\det Dg}\,\dd\lambda
  \leq\int_VF\bigl(g^{-1}(y)\bigr)\,\abs{\det D(g^{-1})(y)}\,\dd y
  = \int_Vf\,\dd\lambda ,
\]
because $F(g^{-1}(y)) = f(y)\abs{\det Dg(g^{-1}(y))}$ and
$Dg(g^{-1}(y))\,D(g^{-1})(y) = I$. So both inequalities are equalities.
For a real $f$, apply this to $\abs f$, $f^+$ and $f^-$.
\end{proof}

\begin{corollary}\label{change-of-variables:cor:sets}
For every Borel set $A\subset U$,
$\lambda(g(A)) = \int_A\abs{\det Dg}\,\dd\lambda$. In particular a
diffeomorphism maps null sets to null sets.
\end{corollary}

\section{How densities transform}\label{change-of-variables:sec:density}

\begin{theorem}[Density of a push-forward]
\label{change-of-variables:thm:pushforward}
Let $\mu$ be a measure on $U$ with density $p$ with respect to Lebesgue
measure. Then $g_\#\mu$ has the density
\[
  q(y) = p\bigl(g^{-1}(y)\bigr)\,\abs{\det D(g^{-1})(y)} ,
  \qquad\text{that is,}\qquad
  q\bigl(g(x)\bigr) = \frac{p(x)}{\abs{\det Dg(x)}} .
\]
\end{theorem}

\begin{proof}
For a Borel set $B\subset V$, apply
\cref{change-of-variables:thm:cov} to the diffeomorphism $g^{-1}$ and
the function $1_{g^{-1}(B)}\,p$ on $U$:
$g_\#\mu(B) = \int_U1_{g^{-1}(B)}\,p\,\dd\lambda
= \int_V1_B(y)\,p(g^{-1}(y))\,\abs{\det D(g^{-1})(y)}\,\dd y$. The second
form follows from $D(g^{-1})(g(x)) = Dg(x)^{-1}$.
\end{proof}

In logarithms, $\ln q(g(x)) = \ln p(x) - \ln\abs{\det Dg(x)}$. For a
composition the determinants multiply, by the chain rule, so the
corrections add:
$\ln\abs{\det D(g_2\circ g_1)(x)}
= \ln\abs{\det Dg_1(x)} + \ln\abs{\det Dg_2(g_1(x))}$.

\begin{keyidea}[A density is not a function]
Under a change of variables $y = g(x)$ a \emph{function} keeps its
values: $h$ becomes $h\circ g^{-1}$. A \emph{density} does not: it is
also divided by $\abs{\det Dg}$, because it is a quantity per unit of
volume and $g$ changes the unit from point to point. What is invariant
is the measure: the probability of a set and of its image are the same
(\cref{change-of-variables:fig:density}). This is the nonlinear form
of the remark that closes \cref{measures:sec:lebesgue}; Part~V will
describe the same behaviour as that of a differential form of top
degree.
\end{keyidea}

\begin{pitfall}[The most probable point depends on the coordinates]
The point where a density is largest is not carried along by a change
of variables (\cref{change-of-variables:fig:density}). So ``the most
probable value'' of a continuous quantity, such as a maximum a
posteriori estimate of a parameter, depends on how the quantity is
parametrized. Statements about probabilities of sets, expectations,
medians of real variables and maxima of likelihoods (which are
functions of the parameter, not densities in it) do not.
\end{pitfall}

\begin{example}[Gaussian measures]\label{change-of-variables:ex:gauss}
By \cref{fubini:ex:gauss} and Tonelli's theorem,
$p(z) = (2\pi)^{-n/2}e^{-\norm z^2/2}$ is a probability density on
$\R^n$, the standard Gaussian. Let $L$ be an invertible matrix and
$m\in\R^n$. By \cref{change-of-variables:thm:pushforward} the law of
$x = m + Lz$ has the density
\[
  q(x) = \frac{e^{-\frac12\norm{L^{-1}(x - m)}^2}}
              {(2\pi)^{n/2}\abs{\det L}}
  = \frac{e^{-\frac12(x - m)^\top\Sigma^{-1}(x - m)}}
         {(2\pi)^{n/2}\sqrt{\det\Sigma}} ,
  \qquad\Sigma = LL^\top .
\]
The normalizing constant of a Gaussian is a Jacobian determinant.
\end{example}

\section{Polar coordinates}\label{change-of-variables:sec:polar}

The map $P(r, \theta) = (r\cos\theta, r\sin\theta)$ is injective on
$(0, \infty)\times(-\pi, \pi)$, of class $C^1$, with
\[
  \det DP(r, \theta)
  = \det\begin{bmatrix}\cos\theta & -r\sin\theta\\
                       \sin\theta & r\cos\theta\end{bmatrix} = r > 0 .
\]
Its image is the plane without the half-line $\{(x, 0) : x\leq0\}$,
which is a null set. So for every measurable $f\geq0$ on $\R^2$, by
\cref{change-of-variables:thm:cov} and Tonelli's theorem,
\[
  \int_{\R^2}f\,\dd\lambda
  = \int_0^\infty\Bigl(\int_{-\pi}^\pi
    f(r\cos\theta, r\sin\theta)\,\dd\theta\Bigr)r\,\dd r .
\]
For $f = e^{-x^2 - y^2}$ this gives
$2\pi\int_0^\infty re^{-r^2}\dd r = \pi$, the square of the Gaussian
integral, in agreement with \cref{fubini:ex:gauss}. The factor $r$ is
the one read off \cref{change-of-variables:fig:polar}.

\section{The volume of a submanifold}
\label{change-of-variables:sec:volume}

Let $M$ be a submanifold of dimension $k$ of $\R^n$ with its usual inner
product, and $\psi:\Omega\to M$ a parametrization of an open piece of
$M$ (\cref{submanifolds:thm:four}), with $\Omega\subset\R^k$ open. The
columns $\partial_1\psi,\dots,\partial_k\psi$ of $D\psi(u)$ are a basis
of the tangent space at $\psi(u)$, with \emph{Gram matrix}
\[
  G(u) = D\psi(u)^\top D\psi(u) , \qquad
  g_{ij} = \partial_i\psi\cdot\partial_j\psi .
\]
By \cref{measures:prop:affine}, in the tangent space the unit cube of
these coordinates has the $k$-dimensional volume $\sqrt{\det G(u)}$. So a
small cell $\dd u$ of $\Omega$ is mapped to a piece of $M$ of volume
about $\sqrt{\det G(u)}\,\dd u$
(\cref{change-of-variables:fig:surface}).

\begin{definition}[Volume in a parametrization]
\label{change-of-variables:def:volume}
For a Borel set $A\subset\psi(\Omega)$,
\[
  \operatorname{vol}_M(A)
  = \int_{\psi^{-1}(A)}\sqrt{\det G(u)}\,\dd u .
\]
\end{definition}

\begin{theorem}\label{change-of-variables:thm:independent}
The number $\operatorname{vol}_M(A)$ does not depend on the
parametrization: if $A$ also lies in the image of a parametrization
$\tilde\psi$, with Gram matrix $\tilde G$, then
$\int_{\tilde\psi^{-1}(A)}\sqrt{\det\tilde G}\,\dd\lambda
= \int_{\psi^{-1}(A)}\sqrt{\det G}\,\dd\lambda$.
\end{theorem}

\begin{proof}
On the common part of the two images, $\tilde\psi = \psi\circ h$ for a
diffeomorphism $h$ between open subsets of $\R^k$
(\cref{submanifolds:cor:transition}). By the chain rule
$D\tilde\psi = (D\psi\circ h)\,Dh$, so
\[
  \tilde G = Dh^\top\,(G\circ h)\,Dh , \qquad
  \sqrt{\det\tilde G} = \abs{\det Dh}\,\sqrt{\det G}\circ h .
\]
Apply \cref{change-of-variables:thm:cov} to $h$ and
$f = 1_{\psi^{-1}(A)}\sqrt{\det G}$, noting that
$f\circ h = 1_{\tilde\psi^{-1}(A)}\sqrt{\det G}\circ h$.
\end{proof}

A submanifold of $\R^n$ is covered by countably many images of
parametrizations, because the open sets of $\R^n$ have a countable
base. Cutting $M$ into disjoint Borel pieces, each inside one image, and
adding their volumes defines a measure $\operatorname{vol}_M$ on the
Borel sets of $M$; by \cref{change-of-variables:thm:independent} it does
not depend on the pieces. By \cref{fubini:lem:density} and the transfer
theorem, for measurable $f\geq0$,
\[
  \int_{\psi(\Omega)}f\,\dd\!\operatorname{vol}_M
  = \int_\Omega f\bigl(\psi(u)\bigr)\sqrt{\det G(u)}\,\dd u .
\]

\begin{example}\label{change-of-variables:ex:volumes}
\begin{enumerate}[nosep]
  \item \emph{Open sets.} For $k = n$, $\psi$ is a diffeomorphism and
        $\sqrt{\det(D\psi^\top D\psi)} = \abs{\det D\psi}$: the volume
        is Lebesgue measure, and the definition is
        \cref{change-of-variables:cor:sets}.
  \item \emph{Curves.} For $k = 1$, $G = \norm{\psi'(t)}^2$ and the
        volume is the length $\int\norm{\psi'(t)}\,\dd t$
        (\cref{riemannian-optimization:def:length}).
  \item \emph{Graphs.} For $\psi(u) = (u, \phi(u))$ with
        $\phi:\Omega\to\R$, $G = I + \nabla\phi\,\nabla\phi^\top$ and
        $\det G = 1 + \norm{\nabla\phi}^2$.
  \item \emph{The sphere.} With longitude $\theta$ and latitude
        $\varphi$,
        $\psi(\theta, \varphi) = (\cos\varphi\cos\theta,
        \cos\varphi\sin\theta, \sin\varphi)$ has
        $G = \diag(\cos^2\varphi, 1)$. The area of the unit sphere is
        $\int_{-\pi}^\pi\int_{-\pi/2}^{\pi/2}\cos\varphi\,
        \dd\varphi\,\dd\theta = 4\pi$; the meridian that the chart
        omits has area $0$.
\end{enumerate}
\end{example}

\begin{remark}[Riemannian volume]\label{change-of-variables:rem:riemann}
Only the Gram matrix entered. On a Riemannian manifold
(\cref{riemannian-optimization:def:metric}) the Gram matrices $G$ and
$\tilde G$ of the metric in two charts are related as in the proof of
\cref{change-of-variables:thm:independent}, so
$\sqrt{\det G(u)}\,\dd u$ defines the same measure in every chart: the
\emph{Riemannian volume}. A metric determines a measure; this is what
\cref{riemannian-optimization:sec:outlook} announced.
\end{remark}

\begin{proposition}[Push-forward by a parametrization]
\label{change-of-variables:prop:injective}
Let $\mu$ be a measure on $\Omega$ with density $p$ with respect to
Lebesgue measure. Then $\psi_\#\mu$ is carried by $M$ and has the
density $p/\sqrt{\det G}$, evaluated at $\psi^{-1}$, with respect to
$\operatorname{vol}_M$.
\end{proposition}

\begin{proof}
$\psi_\#\mu(A) = \int_{\psi^{-1}(A)}p\,\dd\lambda
= \int_{\psi^{-1}(A)}\frac p{\sqrt{\det G}}\sqrt{\det G}\,\dd\lambda$,
which is the integral of $(p/\sqrt{\det G})\circ\psi^{-1}$ over $A$ for
$\operatorname{vol}_M$.
\end{proof}

So the generative model of \cref{measures:fig:generative}, which has no
density in the plane, has one with respect to the length of its curve
when $g$ is a parametrization of it: $p(z)/\norm{g'(z)}$ at the point
$g(z)$.

\section{Spheres, balls and high dimension}
\label{change-of-variables:sec:spheres}

Let $S^{n-1}$ be the unit sphere of $\R^n$, $n\geq2$, and $\sigma$ its
volume measure, the \emph{area}.

\begin{theorem}[Polar coordinates in $\R^n$]
\label{change-of-variables:thm:polar}
For every measurable $f:\R^n\to[0, \infty]$,
\[
  \int_{\R^n}f\,\dd\lambda
  = \int_0^\infty\Bigl(\int_{S^{n-1}}f(r\omega)\,\dd\sigma(\omega)\Bigr)
    r^{n-1}\,\dd r .
\]
\end{theorem}

\begin{proof}
The inverse of the stereographic projection,
$\psi(u) = (2u, \norm u^2 - 1)/(\norm u^2 + 1)$, is a parametrization of
the sphere without the point $e_n$ by $\R^{n-1}$; let $G$ be its Gram
matrix. The map $\Phi(r, u) = r\psi(u)$ is a bijection of
$(0, \infty)\times\R^{n-1}$ onto $\R^n$ without the half-line
$\{te_n : t\geq0\}$, which is a null set. Its matrix is
$D\Phi = \begin{bsmallmatrix}\psi & rD\psi\end{bsmallmatrix}$. Since
$\norm\psi^2 = 1$, differentiating gives $\psi^\top D\psi = 0$, so
\[
  D\Phi^\top D\Phi
  = \begin{bmatrix}1 & 0\\ 0 & r^2G\end{bmatrix} , \qquad
  \abs{\det D\Phi} = \sqrt{\det(D\Phi^\top D\Phi)}
  = r^{n-1}\sqrt{\det G} .
\]
In particular $D\Phi$ is invertible and $\Phi$ is a diffeomorphism. By
\cref{change-of-variables:thm:cov} and Tonelli's theorem,
\[
  \int_{\R^n}f\,\dd\lambda
  = \int_0^\infty\Bigl(\int_{\R^{n-1}}f\bigl(r\psi(u)\bigr)
    \sqrt{\det G(u)}\,\dd u\Bigr)r^{n-1}\,\dd r ,
\]
and the inner integral is $\int_{S^{n-1}}f(r\omega)\,\dd\sigma(\omega)$,
the point $e_n$ having area $0$.
\end{proof}

Let $\Gamma(s) = \int_0^\infty t^{s-1}e^{-t}\,\dd t$ for $s > 0$; an
integration by parts gives $\Gamma(s + 1) = s\Gamma(s)$, so
$\Gamma(n + 1) = n!$, and $\Gamma(\frac12) = \sqrt\pi$ by
\cref{fubini:ex:gauss}.

\begin{corollary}\label{change-of-variables:cor:ball}
The area of the unit sphere and the volume of the ball of radius $R$ in
$\R^n$ are
\[
  \sigma(S^{n-1}) = \frac{2\pi^{n/2}}{\Gamma(\frac n2)} , \qquad
  \lambda\bigl(\{\norm x\leq R\}\bigr)
  = \frac{\sigma(S^{n-1})}n\,R^n
  = \frac{\pi^{n/2}}{\Gamma(\frac n2 + 1)}\,R^n .
\]
\end{corollary}

\begin{proof}
Take $f(x) = e^{-\norm x^2}$ in \cref{change-of-variables:thm:polar}. The
left-hand side is $\pi^{n/2}$, by Tonelli's theorem. The right-hand side
is $\sigma(S^{n-1})\int_0^\infty e^{-r^2}r^{n-1}\,\dd r
= \frac12\sigma(S^{n-1})\Gamma(\frac n2)$, with $t = r^2$. For the ball
take $f = 1_{\{\norm x\leq R\}}$ and use $\int_0^Rr^{n-1}\dd r = R^n/n$.
\end{proof}

\begin{mlthread}[Machine learning: high-dimensional volume]
\begin{itemize}[nosep]
  \item \emph{Balls are all skin.} The fraction of the volume of a ball
        that lies within $(1 - \epsilon)$ of its radius is
        $(1 - \epsilon)^n$. For $n = 100$ and $\epsilon = 0.05$ it is
        less than $1\%$: almost all the volume is in the outer shell.
  \item \emph{Gaussians are bubbles.} By
        \cref{change-of-variables:thm:polar} the radius of a standard
        Gaussian point has the density
        $\sigma(S^{n-1})(2\pi)^{-n/2}\,r^{n-1}e^{-r^2/2}$
        (\cref{change-of-variables:fig:shell}): the factor $r^{n-1}$,
        which comes from the area of the sphere of radius $r$, beats
        the decay of the density up to $r = \sqrt{n - 1}$. The mode of
        the density, the origin, is nowhere near a typical sample. This
        is the pitfall of \cref{change-of-variables:sec:density} in its
        sharpest form.
  \item \emph{Consequences.} In a latent space of dimension $n$ with a
        Gaussian prior, codes have norm about $\sqrt n$; the midpoint
        of two independent codes has norm about $\sqrt{n/2}$ and falls
        inside the bubble, where the decoder has seen almost no
        training codes. Likewise the ball inscribed in a cube occupies
        a vanishing fraction of it
        (\cref{change-of-variables:exr:8}): uniform samples of a cube
        almost never fall near its centre.
\end{itemize}
\end{mlthread}

\section{Densities in machine learning}
\label{change-of-variables:sec:ml}

\begin{mlthread}[Machine learning: normalizing flows]
A normalizing flow is a diffeomorphism $g_\theta$ of $\R^n$ applied to a
simple distribution, usually $z\sim N(0, I)$ with density $p_Z$. By
\cref{change-of-variables:thm:pushforward}, $x = g_\theta(z)$ has the
density
\[
  \ln p_\theta(x) = \ln p_Z(z) - \ln\abs{\det Dg_\theta(z)} ,
  \qquad z = g_\theta^{-1}(x) .
\]
\begin{itemize}[nosep]
  \item The model has an exact likelihood and can be trained by
        maximum likelihood, unlike the models of
        \cref{measures:sec:ml} that have no density.
  \item A determinant of size $n$ costs about $n^3$ operations, so
        flows are built from layers whose Jacobian matrix is
        triangular: the determinant is the product of the diagonal. A
        \emph{coupling layer} splits $x = (x_1, x_2)$ and sets
        $y_1 = x_1$, $y_2 = x_2\odot e^{s(x_1)} + t(x_1)$, for arbitrary
        networks $s$ and $t$. Its Jacobian matrix is block triangular,
        $\ln\abs{\det} = \sum_is_i(x_1)$, and the inverse is explicit
        (\cref{change-of-variables:exr:6}).
  \item Layers compose and their log-determinants add, as noted after
        \cref{change-of-variables:thm:pushforward}.
  \item When the latent dimension is smaller than the data dimension,
        \cref{change-of-variables:prop:injective} replaces the theorem:
        the density lives on a submanifold and the correction is
        $\frac12\ln\det(Dg_\theta^\top Dg_\theta)$.
  \item The differential entropy of the output is that of the input
        plus $\E[\ln\abs{\det Dg_\theta(z)}]$, which generalizes the
        term $\ln\abs a$ of \cref{lp-spaces:exr:8}.
\end{itemize}
\end{mlthread}

\begin{mlthread}[Machine learning: a prior that is the same in every
parametrization]
A ``uniform'' prior density on a parameter $\theta$ is not uniform
after a change of parameter, by
\cref{change-of-variables:thm:pushforward}: flat in a probability $p$
is not flat in $\ln\frac p{1 - p}$. A statistical model carries a
Riemannian metric that no parametrization has to choose, the Fisher
metric of \cref{ch:natural-gradient}. By
\cref{change-of-variables:rem:riemann} its volume
$\sqrt{\det F(\theta)}\,\dd\theta$ is the same measure in every
parametrization. When its total mass is finite, normalizing it gives
the \emph{Jeffreys prior}. For a coin with probability $\theta$ of
heads, $F = 1/(\theta(1 - \theta))$ and the prior is
$\frac1\pi\theta^{-1/2}(1 - \theta)^{-1/2}$: it is the uniform
distribution on the quarter circle of \cref{natural-gradient:fig:arc}
(\cref{change-of-variables:exr:9}).
\end{mlthread}

\section{Measures known by their integrals}
\label{change-of-variables:sec:functionals}

When a push-forward has no density in $\R^n$,
\cref{change-of-variables:prop:injective} gives it a density with
respect to the volume of a submanifold. It needs a submanifold and a
parametrization of it. There is
another way to handle such measures, which needs neither: describe a
measure by the integrals $\int\varphi\,\dd\mu$ of enough functions
$\varphi$. Probability does this whenever it computes $\E[g(X)]$ from the
law of $X$ (the transfer theorem, \cref{lebesgue-integral:thm:transfer}),
physics does it when it writes Dirac's $\delta$, and L.~Schwartz built on
it his theory of distributions.

\begin{proposition}[Integrals of test functions determine a measure]
\label{change-of-variables:prop:testfunctions}
Let $\mu$ and $\nu$ be Borel measures on $\R^n$ that are finite on
bounded sets. If $\int\varphi\,\dd\mu = \int\varphi\,\dd\nu$ for every
continuous $\varphi:\R^n\to\R$ that vanishes outside a bounded set, then
$\mu = \nu$.
\end{proposition}

\begin{proof}
Let $B$ be an open box, a product of open intervals, and
$\varphi_k(x) = \min\bigl(1, k\operatorname{dist}(x, \R^n\setminus
B)\bigr)$. Each $\varphi_k$ is continuous and vanishes outside $B$. For
$x\in B$ the distance is positive, so $\varphi_k(x)$ increases to $1$;
for $x\notin B$ it is $0$. So $\varphi_k$ increases to $1_B$, and by
monotone convergence (\cref{lebesgue-integral:thm:mct})
$\mu(B) = \lim_k\int\varphi_k\,\dd\mu = \lim_k\int\varphi_k\,\dd\nu =
\nu(B)$. The open boxes, with the empty set, form a $\pi$-system: the
intersection of two open boxes is an open box or empty. They generate
$\mathcal B(\R^n)$: every open set $U$ is the union of the countably
many open boxes with rational corners that lie in $U$, since each point
of $U$ is in one of them. Finally $\R^n$ is the increasing union of the
boxes $(-k, k)^n$, of finite measure. \Cref{measures:cor:uniqueness}
concludes.
\end{proof}

\begin{intuition}
A measure is known once its integrals of continuous functions are
known. A density is one way of writing those integrals,
$\int\varphi\,p\,\dd\lambda$, but not the only one, and a measure without
a density is still perfectly described by the map
$\varphi\mapsto\int\varphi\,\dd\mu$.
\end{intuition}

\begin{example}[The next pose of a robot]\label{change-of-variables:ex:robot}
A robot on a lawn has a pose $(x, y, \theta)$: a position and a heading,
kept here as a real number. It starts at $(0, 0, 0)$, turns by
$\phi$ and drives a distance $d$, so that its next pose is
\[
  \psi(d, \phi) = (d\cos\phi,\ d\sin\phi,\ \phi) .
\]
Both commands are executed with independent Gaussian errors:
$(d, \phi)$ has the density $\rho$ of $N(4, 0.4^2)\otimes N(0, 0.03^2)$.
The law of the next pose is the push-forward $\psi_\#(\rho\lambda^2)$.
\begin{enumerate}[nosep]
  \item \emph{It has no density in $\R^3$.} It is carried by the image
        $H = \psi(\R^2)$, a helicoid: the closed set of the points with
        $x\sin\theta - y\cos\theta = 0$. And $\lambda^3(H) = 0$: by Tonelli
        (\cref{fubini:thm:tonelli}) applied to $1_H$,
        $\lambda^3(H) = \int\lambda^2(H_\phi)\,\dd\phi$, where the slice
        $H_\phi = \{(d\cos\phi, d\sin\phi) : d\in\R\}$ at height $\phi$
        is a line of the plane, of area $0$. A density $p$ would give
        $1 = \int_Hp\,\dd\lambda^3 = 0$.
  \item \emph{It has a density with respect to the area of $H$.} The
        map $\psi$ is a parametrization of $H$: the third coordinate of
        $\psi(d, \phi)$ gives $\phi$, and then $d = x\cos\phi +
        y\sin\phi$, so $\psi$ is injective with a continuous inverse;
        its columns $\partial_d\psi = (\cos\phi, \sin\phi, 0)$ and
        $\partial_\phi\psi = (-d\sin\phi, d\cos\phi, 1)$ are orthogonal,
        with $G = \diag(1, 1 + d^2)$. By
        \cref{change-of-variables:prop:injective} the density with
        respect to $\operatorname{vol}_H$ is
        $\rho(d, \phi)/\sqrt{1 + d^2}$ at the point $\psi(d, \phi)$. At
        the commanded pose $(d, \phi) = (4, 0)$ it is
        $\frac{1}{2\pi\times0.4\times0.03}/\sqrt{17} = 13.26/4.12 = 3.22$
        per unit of area. That area is measured with the Euclidean
        metric of $(x, y, \theta)$, which counts a radian as a metre;
        other units would change the number, though not the law.
  \item \emph{As a functional it needs neither.} For every bounded
        measurable $g$ on $\R^3$, by the transfer theorem and
        \cref{fubini:lem:density},
        \[
          \E\bigl[g(\text{next pose})\bigr]
          = \iint g\bigl(\psi(d, \phi)\bigr)\,\rho(d, \phi)\,
            \dd d\,\dd\phi ,
        \]
        an ordinary integral in the plane, and by
        \cref{change-of-variables:prop:testfunctions} these numbers
        determine the law. This is how filters for robot localization
        compute with it: they push samples of $(d, \phi)$ through
        $\psi$.
\end{enumerate}
\end{example}

\paragraph{Distributions.} Schwartz kept only the functional, and allowed
functionals that come from no measure. His test functions are smooth.

\begin{lemma}[A smooth bump]\label{change-of-variables:lem:bump}
The function $\beta$ equal to $e^{-1/(1 - x^2)}$ for $\abs x < 1$ and to
$0$ elsewhere is $C^\infty$ on $\R$, vanishes outside $(-1, 1)$, and
$\beta(0) = e^{-1} > 0$.
\end{lemma}

\begin{proof}
Let $h(t) = e^{-1/t}$ for $t > 0$ and $h(t) = 0$ for $t\leq0$. For
$t > 0$, induction on $k$ gives $h^{(k)}(t) = P_k(1/t)\,e^{-1/t}$ with a
polynomial $P_k$: differentiating $P_k(1/t)e^{-1/t}$ gives
$\bigl(P_k(1/t) - P_k'(1/t)\bigr)t^{-2}e^{-1/t}$. With $s = 1/t$,
$s^me^{-s}\to0$ as $s\to\infty$ for every $m$, so $h^{(k)}(t)\to0$ and
$h^{(k)}(t)/t\to0$ as $t\to0^+$. By induction every $h^{(k)}(0)$ exists
and is $0$: the difference quotient $(h^{(k)}(t) - 0)/t$ tends to $0$
from the right and is $0$ on the left. So $h$ is $C^\infty$, and so is
$\beta(x) = h(1 - x^2)$ by the chain rule.
\end{proof}

\begin{definition}[Distribution]\label{change-of-variables:def:distribution}
Let $\mathcal D(\R^n)$ be the space of $C^\infty$ functions
$\R^n\to\R$ that vanish outside a bounded set, the \emph{test
functions}. A \emph{distribution} is a linear map
$T:\mathcal D(\R^n)\to\R$, written $\varphi\mapsto\langle T,
\varphi\rangle$, such that for every closed ball $K$ there are $C$ and
$m$ with
\[
  \abs{\langle T, \varphi\rangle}
  \leq C\sum_{\abs\alpha\leq m}\sup\abs{\partial^\alpha\varphi}
  \qquad\text{for every $\varphi$ that vanishes outside $K$.}
\]
Here $\alpha = (\alpha_1,\dots,\alpha_n)$ runs over the $n$-tuples of
non-negative integers, $\abs\alpha = \alpha_1 + \dots + \alpha_n$ and
$\partial^\alpha = \partial_1^{\alpha_1}\cdots\partial_n^{\alpha_n}$.
\end{definition}

\begin{example}\label{change-of-variables:ex:distributions}
\begin{enumerate}[nosep]
  \item \emph{Functions.} A function $f$ integrable on bounded sets
        gives $\langle T_f, \varphi\rangle = \int f\varphi\,\dd\lambda$,
        with $m = 0$ and $C = \int_K\abs f\,\dd\lambda$.
  \item \emph{Measures.} A Borel measure finite on bounded sets gives
        $\langle T_\mu, \varphi\rangle = \int\varphi\,\dd\mu$, with
        $m = 0$ and $C = \mu(K)$. The robot's law above is one.
  \item \emph{Dirac.} $\langle\delta_a, \varphi\rangle = \varphi(a)$:
        the measure of mass $1$ at the point $a$, with no density.
  \item \emph{Derivatives.} For $f$ of class $C^1$, integration by parts
        on a bounded interval outside which $\varphi$ vanishes gives
        $\int f'\varphi = -\int f\varphi'$, with no boundary term. This
        suggests the definition
        $\langle T', \varphi\rangle = -\langle T, \varphi'\rangle$ for
        every distribution $T$ on $\R$, which agrees with $T_{f'}$ when
        $f$ is $C^1$. It is a distribution: $\varphi'$ vanishes outside
        $K$ when $\varphi$ does, so if $T$ satisfies the bound with $C$
        and $m$, then $T'$ satisfies it with $C$ and $m + 1$. In particular
        $\langle\delta_a', \varphi\rangle = -\varphi'(a)$, with
        $m = 1$; it comes from no measure
        (\cref{change-of-variables:exr:10}).
\end{enumerate}
\end{example}

Two theorems, quoted here without proof, say where measures sit among
these functionals. A linear map $L$ on the continuous functions that
vanish outside a bounded set, with $L(\varphi)\geq0$ whenever
$\varphi\geq0$, is $\varphi\mapsto\int\varphi\,\dd\mu$ for a unique Borel
measure $\mu$ finite on bounded sets: this is the Riesz representation
theorem (W.~Rudin, \emph{Real and Complex Analysis}, 3rd ed.,
Theorem~2.14), and its uniqueness part is
\cref{change-of-variables:prop:testfunctions}. And a distribution $T$
with $\langle T, \varphi\rangle\geq0$ whenever $\varphi\geq0$ is a
measure (L.~H\"ormander, \emph{The Analysis of Linear Partial
Differential Operators~I}, 2nd ed., Theorem~2.1.7). So the probability
distributions are exactly the positive distributions whose measure has
total mass $1$.

\begin{pitfall}[The delta notation]
Physics and engineering texts write the robot's law of
\cref{change-of-variables:ex:robot}, and push-forwards in general, as if
it had a density:
\[
  p(y) = \int\delta\bigl(y - \psi(z)\bigr)\,\rho(z)\,\dd z .
\]
When the push-forward has no density, as for the robot, this $p$ is not
a function of $y$, and $\delta(y - \psi(z))$ has no value at any point.
The formula means: for every test function $\varphi$,
\[
  \int\varphi(y)\,p(y)\,\dd y
  = \int\Bigl(\int\varphi(y)\,\delta\bigl(y - \psi(z)\bigr)\,\dd y\Bigr)
    \rho(z)\,\dd z
  = \int\varphi\bigl(\psi(z)\bigr)\,\rho(z)\,\dd z ,
\]
which is the transfer theorem, with
$\langle\delta_{\psi(z)}, \varphi\rangle = \varphi(\psi(z))$ in the
middle. A computation with $\delta$ is correct when it can be read in
this way; evaluating $p(y)$, or taking its logarithm, is not, unless the
push-forward has a density, which the formula then gives: for a
diffeomorphism $\psi$ it is the density of
\cref{change-of-variables:thm:pushforward}.
\end{pitfall}

\begin{mlthread}[Machine learning: comparing models through test functions]
The generative models of \cref{measures:sec:ml} whose latent space is
smaller than the data space have no density, so their likelihood
cannot be computed; but by
\cref{change-of-variables:prop:testfunctions} a model is equal to the
data distribution as soon as the two give the same integrals to every
test function. Training methods that compare a model $\mu_\theta$ with
the data $\nu$ use exactly this: they look for a function $\varphi$
in a class $\mathcal F$ on which $\int\varphi\,\dd\mu_\theta$ and
$\int\varphi\,\dd\nu$ differ most, and reduce that difference. With
$\mathcal F$ the functions of Lipschitz constant at most $1$, and for
laws with a finite first moment, $\int\abs y\,\dd\mu(y) < \infty$, the
largest difference is the Wasserstein distance $W_1$ (C.~Villani,
\emph{Optimal Transport, Old and New}, Remark~6.5), the quantity that the
Wasserstein GAN estimates with a neural network as $\varphi$
(M.~Arjovsky, S.~Chintala and L.~Bottou, Wasserstein generative
adversarial networks, ICML, 2017). Both integrals are expectations, so
they are estimated by averages over samples: no density is ever
evaluated. These functions are not test functions, since they need not
vanish outside a bounded set, but a difference of $0$ still forces
$\mu_\theta = \nu$: the functions $\varphi_k$ in the proof of
\cref{change-of-variables:prop:testfunctions} are Lipschitz, with
constant $k$, so the proof goes through with them.
\end{mlthread}

\section{Where the chapter leads}\label{change-of-variables:sec:outlook}

Part~IV is complete: measures, the integral and its limit theorems,
products, spaces of functions, and changes of variables. Two features of
this chapter point to Part~V. First, the absolute value: the theorem
ignores whether $g$ preserves or reverses orientation.
\Cref{ch:multilinear-algebra,ch:differential-forms} introduce objects
that transform with $\det Dg$ itself, differential forms, which can be
integrated over oriented manifolds without any metric. Second, the
divergence: the derivative of $\ln\abs{\det Dg_t}$ along a flow is the
divergence of the velocity field, which governs how densities evolve
in \cref{ch:odes-flows} and what the boundary term is in Stokes'
theorem (\cref{ch:stokes}).

\paragraph{Sources.} The proof of the theorem, through the inequality
$\lambda(g(A))\leq\int_A\abs{\det Dg}$ on cubes and its application to
$g^{-1}$, is that of Loomis and Sternberg, \emph{Advanced Calculus},
\S8.11 (Theorem~11.1 and Lemma~11.1), adapted from contented sets to
Borel sets. Hubbard and Hubbard state the theorem for Riemann integrals
(Theorem~4.10.12) and for the Lebesgue integral (Theorem~4.11.20), treat
polar and spherical coordinates (Propositions~4.10.3 and~4.10.7), and
define the volume of a parametrized manifold by
$\int\sqrt{\det(D\gamma^\top D\gamma)}$ (Definition~5.3.1).
J.~J.~Callahan, \emph{Advanced Calculus: A Geometric View}, Springer,
2010, proves in the plane that the Jacobian determinant is the local
magnification factor of area (his Theorem~9.8: the area of
$\varphi(S)$ is $\iint_S\abs{J_\varphi}$), and gives orientation a
section of its own (\S9.4), where Theorem~9.13 is that a region and its
image have the same orientation exactly when the Jacobian determinant
is positive: the sign that this theorem discards by taking the absolute
value. Coupling
layers are from L.~Dinh, D.~Krueger and Y.~Bengio, NICE: non-linear
independent components estimation, ICLR workshop, 2015, and
L.~Dinh, J.~Sohl-Dickstein
and S.~Bengio, Density estimation using Real NVP, ICLR, 2017; a survey:
G.~Papamakarios et al., Normalizing flows for probabilistic modeling
and inference, \emph{Journal of Machine Learning Research}, 2021. The
invariant prior is from H.~Jeffreys, An invariant form for the prior
probability in estimation problems, \emph{Proceedings of the Royal
Society A}, 1946.

\paragraph{Notebook.} The numerical claims of this chapter are checked
with PyTorch in \notebook{20-change-of-variables.ipynb}.

\begin{chaptersummary}
\begin{itemize}[nosep]
  \item For a $C^1$ diffeomorphism $g:U\to V$,
        $\int_Vf(y)\,\dd y = \int_Uf(g(x))\abs{\det Dg(x)}\,\dd x$, and
        $\lambda(g(A)) = \int_A\abs{\det Dg}\,\dd\lambda$.
  \item If $\mu$ has density $p$, then $g_\#\mu$ has density
        $q(g(x)) = p(x)/\abs{\det Dg(x)}$. Densities are not functions:
        their modes move under a change of variables.
  \item Polar coordinates: $\dd x\,\dd y = r\,\dd r\,\dd\theta$, and in
        $\R^n$, $\dd\lambda = r^{n-1}\,\dd r\,\dd\sigma$.
  \item The volume of a piece of submanifold is
        $\int\sqrt{\det G}\,\dd u$ with $G = D\psi^\top D\psi$,
        whatever the parametrization; a Riemannian metric defines a
        volume in the same way.
  \item $\sigma(S^{n-1}) = 2\pi^{n/2}/\Gamma(\frac n2)$; in high
        dimension the volume of a ball is near its surface and a
        Gaussian sample is at distance about $\sqrt n$ from the mean.
  \item A measure is determined by its integrals of continuous test
        functions, so a law without a density, such as a push-forward
        onto a submanifold, is handled as the functional
        $\varphi\mapsto\int\varphi\,\dd\mu$. Schwartz's distributions are
        such functionals on smooth test functions; the positive ones are
        measures, and $\delta$ is the simplest.
  \item Flows: $\ln p_\theta(x) = \ln p_Z(z) - \ln\abs{\det Dg_\theta(z)}$.
        The measure $\sqrt{\det F}\,\dd\theta$ of the Fisher metric does
        not depend on the parametrization.
\end{itemize}
\end{chaptersummary}

\begin{exercise}\label{change-of-variables:exr:1}
Compute the area of the ellipse $\{x^2/a^2 + y^2/b^2\leq1\}$, and the
volume of the ellipsoid $\{x\in\R^n : x^\top\Sigma^{-1}x\leq1\}$ for a
symmetric positive definite $\Sigma$, in terms of the volume $\omega_n$
of the unit ball.
\solutionlink{change-of-variables:exr:1}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:2}
\begin{enumerate}[nosep]
  \item Compute $\int_{\R^2}(1 + \norm x^2)^{-2}\,\dd x$.
  \item In $\R^n$, for which $a > 0$ is $\norm x^{-a}$ integrable on the
        unit ball? Outside it?
\end{enumerate}
\solutionlink{change-of-variables:exr:2}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:3}
Let $U_1, U_2$ be independent and uniform on $(0, 1)$.
\begin{enumerate}[nosep]
  \item Find the densities of $-\ln U_1$ and of $U_1^2$.
  \item (Box--Muller.) Show that
        $Z_1 = \sqrt{-2\ln U_1}\cos(2\pi U_2)$ and
        $Z_2 = \sqrt{-2\ln U_1}\sin(2\pi U_2)$ are independent standard
        Gaussian variables.
\end{enumerate}
\solutionlink{change-of-variables:exr:3}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:4}
Let $X\sim N(m, s^2)$ and $Y = e^X$. Find the density of $Y$, its mode
and its median, and compare them with the images of the mode and of the
median of $X$.
\solutionlink{change-of-variables:exr:4}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:5}
Let $\Sigma = LL^\top$ with $L$ lower triangular with positive diagonal
(the Cholesky factor). Show that the logarithm of the Gaussian density
of \cref{change-of-variables:ex:gauss} is
\[
  \ln q(x) = -\tfrac12\norm{L^{-1}(x - m)}^2 - \sum_i\ln L_{ii}
  - \tfrac n2\ln(2\pi) ,
\]
and explain why this is how it is computed in practice.
\solutionlink{change-of-variables:exr:5}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:6}
For the coupling layer $y_1 = x_1$, $y_2 = x_2\odot e^{s(x_1)} + t(x_1)$
on $\R^{d}\times\R^{n-d}$:
\begin{enumerate}[nosep]
  \item write the Jacobian matrix in blocks and its determinant;
  \item write the inverse map;
  \item for $n = 2$, $s(x_1) = x_1$, $t(x_1) = x_1^2$ and a standard
        Gaussian input, write the density of the output.
\end{enumerate}
\solutionlink{change-of-variables:exr:6}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:7}

  \item For $n = 3$ the parametrization $\psi$ of the proof of

\begin{exercise}\label{change-of-variables:exr:8}

\begin{exercise}\label{change-of-variables:exr:9}
For the family of coin tosses, $F(\theta) = 1/(\theta(1 - \theta))$ on
$(0, 1)$.
\begin{enumerate}[nosep]
  \item Compute the total mass of $\sqrt{F(\theta)}\,\dd\theta$, and
        relate it to \cref{natural-gradient:prop:sphere}.
  \item Compute the Fisher information in the parameter
        $\eta = \ln\frac\theta{1 - \theta}$ and check that
        $\sqrt{F}\,\dd\theta$ and $\sqrt{F_\eta}\,\dd\eta$ are the same
        measure.
  \item What is the density, in $\eta$, of the prior that is uniform
        in $\theta$?
\end{enumerate}
\solutionlink{change-of-variables:exr:9}
\end{exercise}

\begin{exercise}\label{change-of-variables:exr:10}
\emph{$\delta'$ is not a measure.}
\begin{enumerate}[nosep]
  \item Show that if $\mu$ is a Borel measure finite on bounded sets,
        then $\abs{\int\varphi\,\dd\mu}\leq\mu([-1, 1])\sup\abs\varphi$
        for every continuous $\varphi$ that vanishes outside $[-1, 1]$.
  \item With the bump $\beta$ of \cref{change-of-variables:lem:bump},
        let $\varphi_k(x) = \beta(x)\sin(kx)$. Compute
        $\langle\delta_0', \varphi_k\rangle$ and bound
        $\sup\abs{\varphi_k}$.
  \item Conclude that there is no measure $\mu$ with
        $\langle\delta_0', \varphi\rangle = \int\varphi\,\dd\mu$ for
        every test function $\varphi$.
  \item Show that
        $\langle\delta_0', \varphi\rangle
        = \lim_{\epsilon\to0}\bigl(\varphi(-\epsilon)
        - \varphi(\epsilon)\bigr)/(2\epsilon)$ for every test function
        $\varphi$: $\delta_0'$ is the limit of the charges
        $1/(2\epsilon)$ at $-\epsilon$ and $-1/(2\epsilon)$ at
        $\epsilon$, the dipole of physics.
\end{enumerate}
\hint{The derivative of $\beta(x)\sin(kx)$ at $0$ is $k\beta(0)$.}
\solutionlink{change-of-variables:exr:10}
\end{exercise}
