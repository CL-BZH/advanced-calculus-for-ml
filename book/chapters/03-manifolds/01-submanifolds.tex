\chapter{Submanifolds of an Affine Space}
\label{ch:submanifolds}
\chaptersubtitle{%
  Graphs, level sets and parametrizations}

\begin{chapterplan}
\begin{itemize}[nosep]
  \item Three descriptions of a submanifold (locally a graph, a level set of a
        submersion, the image of a parametrization) and their equivalence
        through the implicit function theorem.
  \item The tangent space $T_pM$, a subspace of $V$ that varies with $p$, and
        the tangent flat $p + T_pM$.
  \item Examples: spheres, the orthogonal group, Stiefel manifolds, matrices of
        fixed rank, the open simplex.
  \item Differentiable maps between submanifolds and their differentials.
\end{itemize}

  \item Low-rank parametrizations.
  \item The manifold hypothesis for data.
\end{itemize}
\end{chapterplan}
