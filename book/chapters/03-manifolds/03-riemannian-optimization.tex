\chaptersubtitle{%
  Gradient descent without $\theta + v$}

\begin{chapterplan}
\begin{itemize}[nosep]
  \item Riemannian metrics; the Riemannian gradient; for a submanifold of a

  \item Retractions and the exponential map; Riemannian gradient descent and its
        convergence.
  \item Geodesics and curvature, and what ``flat'' means precisely.
  \item Worked examples: the sphere (Rayleigh quotient, PCA) and the Stiefel
        manifold.
\end{itemize}
\smallskip\noindent\textbf{Machine learning threads.}
\begin{itemize}[nosep]
  \item Orthogonality constraints and weight normalization.
  \item Hyperbolic embeddings.
\end{itemize}
\end{chapterplan}
