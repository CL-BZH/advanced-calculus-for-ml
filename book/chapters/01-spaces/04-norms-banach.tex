\begin{notechapter}
\chapter{Norms, Topology and Banach Spaces}
\label{ch:norms-banach}
\chaptersubtitle{%
  Convergence, continuity and bounded operators}

\begin{chapterplan}
\begin{itemize}[nosep]
  \item Norms and distances; open and closed sets; continuity; compactness;
        completeness.
  \item All norms on a finite-dimensional space are equivalent; this fails in
        infinite dimension.
  \item Bounded linear maps and their equivalent characterizations; the operator
        norm; $L(V, W)$ is a Banach space when $W$ is.
  \item Extension of a bounded operator from a dense subspace.
  \item Function spaces: $C^0([a,b])$ with the sup norm; Hilbert spaces;
        projection onto a closed subspace and the Riesz theorem in infinite
        dimension.
  \item The contraction mapping principle, used in
        \cref{ch:contractions-newton,ch:inverse-implicit,ch:odes-flows}.
\end{itemize}
\smallskip\noindent\textbf{Machine learning threads.}
\begin{itemize}[nosep]
  \item Lipschitz constants of networks and spectral normalization.
  \item Learning as a search in a space of functions.
\end{itemize}
\end{chapterplan}
\end{notechapter}
